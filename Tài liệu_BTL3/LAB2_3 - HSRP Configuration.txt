!Cấu hình cho router CoreRT1
en
conf t
hostname CoreRT1
int g0/0
	ip add 172.16.10.2 255.255.255.0
	no shut
int g0/1
	no shutdown
int g0/1.100
	description VLAN 100
	encapsulation dot1q 100
	ip add 192.168.100.1 255.255.255.0
	standby 1 ip 192.168.100.100 
	! đây là Virtual IP cho group 1, các trạm dùng địa chỉ này làm Default Gateway
	standby 1 priority 105 ! mặc định là 100; cao hơn nên làm active
	standby 1 track g0/0 
	! khi cổng này down thì priority của router bị giảm xuống 10 (mặc định)
	standby 1 preempt
	! cho phép router này trở thành active router khi nó có priority cao hơn các router cấu hình HSRP khác trong nhóm standby
router ospf 1
	network 172.16.10.0 0.0.0.255 area 0
	network 192.168.100.0 0.0.0.255 area 0
